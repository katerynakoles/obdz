# Показник із дашборду роботи №3: виручка за вересень 2026 року (2540.00 грн)
# Правило ключа: <metric>:<period> від загального до конкретного через двокрапку -> revenue:2026-09

# Команда 1. Записати значення показника
SET revenue:2026-09 2540.00
# OK

# Команда 2. Прочитати значення
GET revenue:2026-09
# "2540.00"

# Команда 3. Перевірити час життя ключа без заданого строку (-1)
TTL revenue:2026-09
# (integer) -1

# Команда 4. Задати час життя ключа (25 секунд для швидкого досліду зникнення)
EXPIRE revenue:2026-09 25
# (integer) 1

# Команда 5. Перевірити залишок часу життя (додатне число)
TTL revenue:2026-09
# (integer) 22

# Команда 6. Перевірити наявність ключа
EXISTS revenue:2026-09
# (integer) 1

# Команда 7. Дочекатися зникнення ключа (через 25+ секунд) і прочитати знову
GET revenue:2026-09
# (nil)

# Команда 8. Перевірити час життя неіснуючого ключа (-2)
TTL revenue:2026-09
# (integer) -2

# Команда 9. Записати значення одразу з часом життя (EX 25)
SET revenue:2026-09 2540.00 EX 25
# OK

# Команда 10. Порахувати щось лічильником (кількість переглядів звіту за день)
INCR dashboard:views:2026-11-28
# (integer) 1

# Шаблон «спершу кеш, потім база» (для показника revenue:2026-10 = 2060.00 грн):
# 1. Промах кеша:
GET revenue:2026-10
# (nil)
# 2. Запит до PostgreSQL (схема «зірка» роботи №3):
# SELECT sum(line_revenue) FROM fact_order_items f JOIN dim_dates d ON d.date_key = f.date_key WHERE d.month_label = '2026-10'; -> 2060.00
# 3. Запис у кеш із часом життя:
SET revenue:2026-10 2060.00 EX 25
# OK
# 4. Повторне читання з кеша (попадання):
GET revenue:2026-10
# "2060.00"
